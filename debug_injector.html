🔧 FieldMind 实时调试工具

调试脚本

复制以下代码，在 FieldMind 应用中按 Command+Option+I 打开开发者工具，粘贴到 Console 并回车：

// FieldMind 实时调试脚本
console.clear();
console.log('%c🔍 FieldMind 调试开始', 'color: #4ec9b0; font-size: 16px; font-weight: bold;');

// 1. 检查函数是否存在
console.log('\n📋 检查关键函数:');
console.log('  goStep:', typeof goStep);
console.log('  handleFileSelect:', typeof handleFileSelect);
console.log('  handleFileSelect_newProject:', typeof handleFileSelect_newProject);
console.log('  showPage:', typeof showPage);

// 2. 测试 goStep 函数
console.log('\n🧪 测试 goStep(2):');
try {
    goStep(2);
    console.log('%c  ✅ goStep(2) 执行成功', 'color: #4ec9b0;');

    // 检查步骤2是否显示
    const step2 = document.getElementById('step-2');
    if (step2) {
        const isVisible = step2.classList.contains('active');
        console.log('  step-2 元素存在:', true);
        console.log('  step-2 可见:', isVisible);
        console.log('  step-2 display:', window.getComputedStyle(step2).display);
    } else {
        console.log('%c  ❌ step-2 元素不存在', 'color: #f48771;');
    }
} catch (error) {
    console.log('%c  ❌ goStep(2) 执行失败:', 'color: #f48771;', error);
}

// 3. 检查按钮
console.log('\n🔘 检查按钮:');
const buttons = document.querySelectorAll('button[onclick*="goStep(2)"]');
console.log('  找到 goStep(2) 按钮:', buttons.length, '个');
buttons.forEach((btn, i) => {
    console.log(`  按钮${i+1}:`, btn.textContent.trim());
    console.log('    可见:', window.getComputedStyle(btn).display !== 'none');
    console.log('    z-index:', window.getComputedStyle(btn).zIndex);
    console.log('    pointer-events:', window.getComputedStyle(btn).pointerEvents);
});

// 4. 添加点击监听器
console.log('\n🎯 添加调试监听器:');
buttons.forEach((btn, i) => {
    btn.addEventListener('click', function(e) {
        console.log(`%c🖱️  按钮${i+1}被点击了!`, 'color: #dcdcaa; font-size: 14px;');
        console.log('  事件对象:', e);
        console.log('  目标元素:', e.target);
    }, true); // 使用捕获阶段
});
console.log('  ✅ 监听器已添加');

// 5. 检查是否有遮罩层
console.log('\n🛡️  检查遮罩层:');
const overlays = document.querySelectorAll('[class*="overlay"], [class*="modal"]');
overlays.forEach((overlay, i) => {
    const style = window.getComputedStyle(overlay);
    if (style.display !== 'none' && style.position === 'fixed') {
        console.log(`%c  ⚠️  遮罩层${i+1}可能阻挡点击:`, 'color: #dcdcaa;');
        console.log('    class:', overlay.className);
        console.log('    z-index:', style.zIndex);
        console.log('    pointer-events:', style.pointerEvents);
    }
});

// 6. 强制测试点击
console.log('\n🚀 提供手动测试函数:');
window.testGoStep2 = function() {
    console.log('%c手动执行 goStep(2)', 'color: #4ec9b0; font-size: 14px;');
    goStep(2);
    setTimeout(() => {
        const step2 = document.getElementById('step-2');
        if (step2 && step2.classList.contains('active')) {
            console.log('%c✅ 第2步已显示!', 'color: #4ec9b0; font-size: 14px;');
        } else {
            console.log('%c❌ 第2步未显示', 'color: #f48771; font-size: 14px;');
        }
    }, 100);
};

console.log('  输入 testGoStep2() 可手动测试跳转');
console.log('\n%c✨ 调试脚本加载完成', 'color: #4ec9b0; font-size: 16px; font-weight: bold;');
console.log('现在点击"下一步：导入资料"按钮，观察控制台输出...');

快速测试

如果你能打开开发者工具，依次输入这些命令测试：

// 1. 测试基本功能
goStep(2)

// 2. 检查元素
document.getElementById('step-2')

// 3. 强制显示
document.getElementById('step-2').classList.add('active')

// 4. 测试文件选择
handleFileSelect_newProject({target: {files: []}})

如果还是不行

如果上述调试都没有效果，可能需要：

  1. 检查浏览器控制台是否有 JavaScript 错误（红色）
  2. 尝试用 Safari 或 Chrome 直接打开 index.html 文件
  3. 或者我直接创建一个简化的工作版本
